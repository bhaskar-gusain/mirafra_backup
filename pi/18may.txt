#include <linux/module.h>

#include <linux/kernel.h>

#include <linux/init.h>

#include <linux/gpio.h>
 
#define GPIO_LED 17
 
static int __init gpio_driver_init(void)

{

    int ret;
 
    pr_info("GPIO Driver Init\n");
 
    /* Request GPIO */

    ret = gpio_request(GPIO_LED, "GPIO_LED");

    if (ret) {

        pr_err("GPIO Request Failed\n");

        return ret;

    }
 
    /* Set GPIO as output */

    ret = gpio_direction_output(GPIO_LED, 1);

    if (ret) {

        pr_err("GPIO Direction Failed\n");

        gpio_free(GPIO_LED);

        return ret;

    }
 
    /* Turn LED ON */

    gpio_set_value(GPIO_LED, 1);
 
    pr_info("LED ON GPIO %d\n", GPIO_LED);
 
    return 0;

}
 
static void __exit gpio_driver_exit(void)

{

    /* Turn LED OFF */

    gpio_set_value(GPIO_LED, 0);
 
    /* Free GPIO */

    gpio_free(GPIO_LED);
 
    pr_info("GPIO Driver Exit\n");

}
 
module_init(gpio_driver_init);

module_exit(gpio_driver_exit);
 
MODULE_LICENSE("GPL");

MODULE_AUTHOR("Prashant");

MODULE_DESCRIPTION("Basic GPIO Driver for Raspberry Pi 4B");
 
//-------------------------------------------

obj-m += gpio_driver.o
KDIR := /lib/modules/$(shell uname -r)/build

all:
  make -C $(KDIR) M=$(PWD) modules

  make -C $(KDIR) M=$(PWD) clean
